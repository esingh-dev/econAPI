Plug into Any Stack in Under Two Minutes

# pip install requests import requests BASE = "https://econapi.onrender.com" # or http://127.0.0.1:8000 locally # 1. BSM Greeks with Finite-Difference Validation bsm = requests.post(f"{BASE}/v1/derivatives/bsm", json={ "S": 100.0, "K": 100.0, "T": 1.0, "r": 0.05, "sigma": 0.20, "option_type": "call", "validate_greeks": True }).json() print(f"BSM Price: ${bsm['price']:.4f} | Delta: {bsm['delta']:.4f} | Gamma: {bsm['gamma']:.4f}") # 2. Nelson-Siegel-Svensson 6-Parameter Yield Curve nss = requests.post(f"{BASE}/v1/fixed-income/nss", json={ "tenors": [0.25, 0.5, 1.0, 2.0, 5.0, 10.0, 30.0], "yields": [0.052, 0.050, 0.048, 0.045, 0.042, 0.041, 0.044] }).json() print(f"NSS RMSE: {nss['rmse']:.6f} | Converged: {nss['converged']}")
