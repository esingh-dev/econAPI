# pip install requests ─── No other deps needed import requests BASE = "https://econapi.onrender.com" # or http://127.0.0.1:8000 locally HEADERS = {"Content-Type": "application/json"} # add X-RapidAPI-Key for RapidAPI # ── 1. Black-Scholes-Merton: full analytical Greeks ──────────────── bsm = requests.post(f"{BASE}/v1/derivatives/bsm", headers=HEADERS, json={ "S": 100.0, "K": 100.0, "T": 1.0, "r": 0.05, "sigma": 0.20, "option_type": "call" }).json() print(f"Price=${bsm['price']:.4f} Δ={bsm['delta']:.4f} Γ={bsm['gamma']:.4f}") print(f"𝒱={bsm['vega']:.2f} Θ={bsm['theta']/365:.4f}/day ρ={bsm['rho']:.2f}") # ── 2. Heston Stochastic Volatility ──────────────────────────────── heston = requests.post(f"{BASE}/v1/derivatives/heston", headers=HEADERS, json={ "S": 100, "K": 100, "T": 1.0, "r": 0.05, "v0": 0.04, "kappa": 2.0, "theta": 0.04, "xi": 0.30, "rho": -0.70, "option_type": "call" }).json() print(f"Heston Price=${heston['price']:.4f} IV={heston['implied_volatility']*100:.2f}%") # ── 3. Hierarchical Risk Parity ──────────────────────────────────── hrp = requests.post(f"{BASE}/v1/portfolio/hrp", headers=HEADERS, json={ "returns": [ [ 0.01, -0.005, 0.002, 0.008], [-0.015, 0.008, 0.004, -0.012], [ 0.008, -0.002, -0.003, 0.006], ], "asset_names": ["SPY", "TLT", "GLD", "VNQ"] }).json() print("HRP Weights:", hrp["weights"], " Sharpe:", hrp["sharpe_ratio"]) # ── 4. EVT Tail Risk ─────────────────────────────────────────────── evt = requests.post(f"{BASE}/v1/risk/evt-cvar", headers=HEADERS, json={ "returns": [-0.02, 0.01, 0.03, -0.05, -0.08, 0.02, # …min 50 obs -0.065, -0.078, -0.092, -0.115, -0.142], "confidence_level": 0.99, "threshold_quantile": 0.95 }).json() print(f"VaR={evt['var']*100:.2f}% CVaR={evt['cvar']*100:.2f}%") # ── 5. Nelson-Siegel Yield Curve ─────────────────────────────────── ns = requests.post(f"{BASE}/v1/macro/nelson-siegel", headers=HEADERS, json={ "tenors": [0.25, 0.5, 1.0, 2.0, 3.0, 5.0, 7.0, 10.0, 20.0, 30.0], "yields": [5.25, 5.10, 4.85, 4.50, 4.35, 4.20, 4.15, 4.10, 4.30, 4.40] }).json() print(f"β₀={ns['beta_0']:.3f} β₁={ns['beta_1']:.3f} β₂={ns['beta_2']:.3f} RMSE={ns['rmse']:.6f}%")

// TypeScript-compatible — works in Node.js, Deno, Bun, browser const BASE = "https://econapi.onrender.com"; async function econ<T>(endpoint: string, body: object): Promise<T> { const r = await fetch(`${BASE}${endpoint}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body), }); if (!r.ok) throw new Error(`econAPI ${r.status}: ${await r.text()}`); return r.json(); } // 1. BSM Greeks const { price, delta, gamma, vega, theta, rho } = await econ("/v1/derivatives/bsm", { S: 100, K: 100, T: 1.0, r: 0.05, sigma: 0.20, option_type: "call" }); console.log(`Price: $${price.toFixed(4)} | Δ ${delta.toFixed(4)} | Γ ${gamma.toFixed(4)}`); // 2. Heston — async Fourier quadrature (~200–400ms) const h = await econ("/v1/derivatives/heston", { S:100, K:100, T:1.0, r:0.05, v0:0.04, kappa:2.0, theta:0.04, xi:0.30, rho:-0.70, option_type:"call" }); console.log(`Heston IV: ${(h.implied_volatility*100).toFixed(2)}%`); // 3. HRP — parallel with Promise.all for speed const [spy, tech] = await Promise.all([ econ("/v1/portfolio/hrp", { returns: [[0.01,-0.005],[-0.015,0.008],[0.008,-0.002]], asset_names: ["SPY", "TLT"] }), econ("/v1/portfolio/hrp", { returns: [[0.02,0.015,0.035],[-0.018,-0.012,-0.04]], asset_names: ["AAPL", "MSFT", "NVDA"] }), ]); console.log("SPY-TLT:", spy.weights, "| Tech:", tech.weights);
